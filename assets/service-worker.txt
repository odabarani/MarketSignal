/* Research and Dash responses must always come from the server. */
const CACHE = 'marketsignal-offline-__REVISION__';
const LANDING = '/offline';
self.addEventListener('install', event => {
  event.waitUntil(caches.open(CACHE).then(cache => cache.addAll([
    LANDING, '/assets/icons/app-192.png', '/assets/icons/app-512.png'
  ])));
});
self.addEventListener('activate', event => {
  event.waitUntil(caches.keys().then(keys => Promise.all(
    keys.filter(key => key.startsWith('marketsignal-offline-') && key !== CACHE)
      .map(key => caches.delete(key))
  )).then(() => self.clients.claim()));
});
self.addEventListener('fetch', event => {
  if (event.request.method !== 'GET') return;
  const url = new URL(event.request.url);
  if (url.origin !== self.location.origin) return;
  if (event.request.mode === 'navigate' && (url.pathname === '/' || url.pathname === '/offline')) {
    event.respondWith(fetch(event.request).catch(() => caches.match(LANDING)));
  }
});
